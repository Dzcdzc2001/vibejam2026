# 边境远征 MVP 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现边境远征 MVP — 主菜单、主世界地图、火山箱庭区域、回合制战斗(含QTE弹反)、主城铁匠铺/图书馆、图鉴界面、存档系统，全部使用色块占位。

**Tech Stack:** Godot 4.6, GDScript (静态类型), Jolt Physics, Forward Plus 渲染器

在 Godot 编辑器中：
1. 打开项目 `bianjingyuanzheng/project.godot`
2. 在 FileSystem 面板右键 `assets/placeholder/` → New Resource → GradientTexture1D
3. 保存为 `tile_16x16.tres` (16×16 纯色灰度方块)

**替代方案** — 通过代码生成占位纹理：

```gdscript
# 在 Godot 编辑器中临时执行 (Script Editor → File → Run)
# 或创建一个 @tool 脚本运行

extends Node

func _ready():
    # 生成地形色块 64×64 精灵表 (4行×4列 = 16种地形色)
    var terrain_colors = [
        Color.GRAY, Color.DIM_GRAY, Color.SADDLE_BROWN, Color.RED,
        Color.DARK_RED, Color.ORANGE_RED, Color.YELLOW, Color.BLACK,
        Color.WHITE, Color.LIGHT_BLUE, Color.SKY_BLUE, Color.CYAN,
        Color.GREEN, Color.LIME_GREEN, Color.PURPLE, Color.MAGENTA
    ]
    var img = Image.create(256, 256, false, Image.FORMAT_RGBA8)
    for row in 4:
        for col in 4:
            var color = terrain_colors[row * 4 + col]
            for x in range(col * 64, (col + 1) * 64):
                for y in range(row * 64, (row + 1) * 64):
                    img.set_pixel(x, y, color)
    img.save_png("res://assets/placeholder/tile_16x16.png")
    print("Terrain placeholder generated.")

    # 生成角色色块 16×16 精灵
    var actor_colors = {
        "player": Color.BLUE,
        "monster_normal": Color.ORANGE_RED,
        "monster_elite": Color.ORANGE,
        "monster_boss": Color.PURPLE
    }
    var actor_img = Image.create(64, 64, false, Image.FORMAT_RGBA8)
    var idx = 0
    for key in actor_colors:
        var col_offset = (idx % 4) * 16
        var row_offset = (idx / 4) * 16
        for x in range(16):
            for y in range(16):
                actor_img.set_pixel(col_offset + x, row_offset + y, actor_colors[key])
        idx += 1
    actor_img.save_png("res://assets/placeholder/actor_16x16.png")

    # 生成物品图标色块
    var item_colors = [Color.GREEN, Color.BLUE, Color.MAGENTA, Color.ORANGE, Color.YELLOW, Color.GOLD]
    var icon_img = Image.create(96, 16, false, Image.FORMAT_RGBA8)
    for i in range(6):
        for x in range(i * 16, (i + 1) * 16):
            for y in range(16):
                icon_img.set_pixel(x, y, item_colors[i])
    icon_img.save_png("res://assets/placeholder/icon_16x16.png")
    print("All placeholders generated.")
```

- [ ] **Step 1: 编写 SaveManager**

```gdscript
# scripts/autoload/save_manager.gd
extends Node

const SAVE_DIR := "user://saves/"
const MAX_SLOTS := 3

func save_game(slot: int, data: Dictionary) -> bool:
    if slot < 0 or slot >= MAX_SLOTS:
        return false
    
    DirAccess.make_dir_recursive_absolute(SAVE_DIR)
    var file := FileAccess.open(SAVE_DIR + "save_%d.json" % slot, FileAccess.WRITE)
    if file == null:
        return false
    
    data["save_version"] = 1
    data["timestamp"] = Time.get_unix_time_from_system()
    file.store_string(JSON.stringify(data, "\t"))
    file.close()
    
    EventBus.game_saved.emit(slot)
    return true

func load_game(slot: int) -> Dictionary:
    var path := SAVE_DIR + "save_%d.json" % slot
    if not FileAccess.file_exists(path):
        return {}
    
    var file := FileAccess.open(path, FileAccess.READ)
    if file == null:
        return {}
    
    var json := JSON.new()
    var err := json.parse(file.get_as_text())
    file.close()
    
    if err != OK:
        return {}
    
    var data: Dictionary = json.get_data()
    EventBus.game_loaded.emit(slot)
    return data

func delete_save(slot: int) -> void:
    var path := SAVE_DIR + "save_%d.json" % slot
    if FileAccess.file_exists(path):
        DirAccess.remove_absolute(path)

func save_exists(slot: int) -> bool:
    return FileAccess.file_exists(SAVE_DIR + "save_%d.json" % slot)

func get_save_meta(slot: int) -> Dictionary:
    var data := load_game(slot)
    if data.is_empty():
        return {}
    return {
        "level": data.get("player", {}).get("level", 1),
        "play_time": data.get("play_time_seconds", 0),
        "scene": data.get("current_scene", ""),
        "timestamp": data.get("timestamp", 0)
    }
```

```gdscript
    if inventory.has(item_id):
        inventory[item_id] += count
    else:
        inventory[item_id] = count
    EventBus.item_obtained.emit(item_id, count)

func has_item(item_id: String, count: int = 1) -> bool:
    return inventory.get(item_id, 0) >= count

func remove_item(item_id: String, count: int) -> bool:
    if not has_item(item_id, count):

    inventory[item_id] -= count
    if inventory[item_id] <= 0:

        exp -= required
        level += 1
        free_points += 3
        base_hp += 20
        base_spd += 2
        EventBus.player_leveled_up.emit(level)

        if count % 5 == 0:
```

### Task 17: 实现怪物生成器 MonsterSpawner

```gdscript
# scripts/scenes/monster_spawner.gd
class_name MonsterSpawner
extends Area2D

@export var spawn_radius: float = 150.0
@export var max_monsters: int = 3
@export var respawn_time: float = 30.0  # 秒, 怪物被击败后重生间隔

var region_config: RegionConfig = null
var current_weather: String = "clear"
var active_monsters: Array[Node2D] = []
var _respawn_timers: Dictionary = {}  # {monster_id: time_remaining}

func initialize(config: RegionConfig, weather: String) -> void:
    region_config = config
    current_weather = weather
    _spawn_initial_monsters()

func _spawn_initial_monsters() -> void:
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    for i in range(max_monsters):
        _spawn_one_monster(rng)

func _spawn_one_monster(rng: RandomNumberGenerator) -> void:
    if region_config == null:
        return
    
    # 从怪物池随机选择
    var pool := region_config.monster_pool.duplicate()
    var chosen_id := pool[rng.randi_range(0, pool.size() - 1)]
    
    # 加载 MonsterData
    var monster_res := load("res://resources/monsters/%s.tres" % chosen_id) as MonsterData
    if monster_res == null:
        return
    
    # 创建地图上的怪物节点
    var monster_node := Area2D.new()
    var sprite := Sprite2D.new()
    sprite.texture = load("res://assets/placeholder/actor_16x16.png")
    # 根据类型选颜色: 普通=红, 精英=橙, BOSS=紫
    match monster_res.monster_type:
        1: sprite.modulate = Color.ORANGE
        2: sprite.modulate = Color.PURPLE
        _: sprite.modulate = Color.ORANGE_RED
    
    monster_node.add_child(sprite)
    
    var collision := CollisionShape2D.new()
    var rect := RectangleShape2D.new()
    rect.size = Vector2(16, 16)
    collision.shape = rect
    monster_node.add_child(collision)
    
    # 随机位置
    var angle := rng.randf_range(0, TAU)
    var distance := rng.randf_range(0, spawn_radius)
    monster_node.position = position + Vector2(cos(angle), sin(angle)) * distance
    
    # 添加 AI 脚本
    var ai_script := load("res://scripts/scenes/monster_ai.gd")
    monster_node.set_script(ai_script)
    monster_node.set("monster_data", monster_res)
    
    # 连接触碰信号
    monster_node.body_entered.connect(_on_monster_touched.bind(monster_node))
    
    get_parent().add_child(monster_node)
    active_monsters.append(monster_node)

func _on_monster_touched(body: Node2D, monster_node: Node2D) -> void:
```

```gdscript
# scripts/scenes/monster_ai.gd
class_name MonsterAI
extends Area2D

enum State { WANDER, ALERT, CHASE, FLEE }

var monster_data: MonsterData = null
var state: int = State.WANDER
var _wander_timer: float = 0.0
var _wander_interval: float = 3.0
var _wander_direction: Vector2 = Vector2.ZERO
var _move_speed: float = 50.0
var _chase_speed: float = 120.0

@export var alert_radius: float = 48.0
@export var chase_radius: float = 32.0
@export var wander_radius: float = 150.0
@export var spawn_origin: Vector2 = Vector2.ZERO

func _ready() -> void:
    spawn_origin = position
    _pick_new_wander_direction()

func _physics_process(delta: float) -> void:
    var player := _find_player()
    if player == null:
        state = State.WANDER
    else:
        var dist := position.distance_to(player.position)
        if state == State.FLEE:
            pass  # 逃离逻辑
        elif dist < chase_radius:
            state = State.CHASE
        elif dist < alert_radius:
            state = State.ALERT
        else:
            state = State.WANDER
    
    match state:
        State.WANDER:
            _do_wander(delta)
        State.ALERT:
            pass  # 原地警觉
        State.CHASE:
            _do_chase(player, delta)
        State.FLEE:
            _do_flee(delta)

func _do_wander(delta: float) -> void:
    _wander_timer -= delta
    if _wander_timer <= 0:
        _pick_new_wander_direction()
    
    var new_pos := position + _wander_direction * _move_speed * delta
    if new_pos.distance_to(spawn_origin) <= wander_radius:
        position = new_pos
    else:
        _wander_direction = -_wander_direction

func _do_chase(player: Node2D, delta: float) -> void:
    var dir := position.direction_to(player.position)
    position += dir * _chase_speed * delta

func _do_flee(delta: float) -> void:
    var dir := position.direction_to(spawn_origin)
    position += dir * _move_speed * delta

func _pick_new_wander_direction() -> void:
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    var angle := rng.randf_range(0, TAU)
    _wander_direction = Vector2(cos(angle), sin(angle))
    _wander_timer = rng.randf_range(2.0, 4.0)
    _wander_interval = rng.randf_range(3.0, 5.0)

func _find_player() -> Node2D:
    var players := get_tree().get_nodes_in_group("player_character")
    if players.is_empty():
        return null
    return players[0]
```

```gdscript
# scripts/scenes/skill_parser.gd
class_name SkillParser
extends RefCounted

var keyword_db: SkillKeywordDB = null

func _init() -> void:
    keyword_db = load("res://resources/keywords/skill_keyword_db.tres")

func parse_creativity(input_text: String, region_type: String, weapon_type: String, monster_element: String) -> int:
    if input_text.is_empty():
        return 20
    
    # 0. 检查彩蛋词
    for word in keyword_db.easter_egg_words:
        if input_text.contains(word):
            return keyword_db.easter_egg_words[word]
    
    var score: int = 20  # 基础分
    
    # 1. 主题契合度 (40分)
    score += _score_theme_fit(input_text, region_type, weapon_type, monster_element)
    
    # 2. 创意表达 (30分)
    score += _score_creativity(input_text)
    
    # 3. 字面分析 (20分)
    score += _score_literal(input_text)
    
    # 4. 随机偏差 ±5
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    score += rng.randi_range(-5, 5)
    
    return clampi(score, 0, 100)

func _score_theme_fit(text: String, region: String, weapon: String, element: String) -> int:
    var score := 0
    var region_kw := keyword_db.get_region_keywords(region)
    var weapon_kw := keyword_db.get_weapon_keywords(weapon)
    var element_kw := keyword_db.get_elemental_keywords(element)
    
    for kw in region_kw:
        if text.contains(kw):
            score += 15
            break
    for kw in weapon_kw:
        if text.contains(kw):
            score += 10
            break
    for kw in element_kw:
        if text.contains(kw):
            score += 15
            break
    
    return min(score, 40)

func _score_creativity(text: String) -> int:
    var score := 10  # 基础分
    if text.length() >= 2: score += 5
    if text.length() >= 4: score += 5
    # 包含特殊字符（如 "·" "之" "的"）视为有组合词
    if text.contains("之"): score += 5
    if text.contains("的"): score += 3
    # 非纯ASCII词
    var has_chinese := false
    for c in text:
        if c.unicode_at(0) > 127:
            has_chinese = true
            break
    if has_chinese: score += 5
    return min(score, 30)

func _score_literal(text: String) -> int:
    var score := 10
    var action_words := ["斩", "击", "破", "裂", "刺", "射", "冲", "劈", "砍", "碎", "崩", "震", "突", "袭", "轰"]
    for w in action_words:
        if text.contains(w):
            score += 5
            break
    var imagery_words := ["星", "冰", "焰", "火", "光", "影", "暗", "风", "雷", "月", "日", "天", "地", "海", "山"]
    for w in imagery_words:
        if text.contains(w):
            score += 5
            break
    return min(score, 20)
```

```gdscript
# scripts/scenes/damage_calculator.gd
class_name DamageCalculator
extends RefCounted

# 常量
const BASE_SKILL_MULTIPLIER: float = 1.0
const REGION_BONUS_DEFAULT: float = 1.0
const WEATHER_BONUS_DEFAULT: float = 1.0
const STAT_MULTIPLIER: float = 10.0

static func calc_player_damage(weapon: WeaponData, creativity: int, monster_def: int, region: String, weather: String) -> int:
    # 武器基础ATK取中间值
    var weapon_atk: float = (weapon.base_atk_min + weapon.base_atk_max) / 2.0 if weapon else 50.0
    
    # 武器系数
    var weapon_mult: float = 1.0
    if weapon:
        weapon_mult = 1.0 + (weapon.level - 1) * 0.05
        weapon_mult += weapon.element_bonus - 1.0  # 元素加成
    
    # 技能乘数
    var skill_mult: float = BASE_SKILL_MULTIPLIER
    
    # 地域加成
    var region_mult: float = REGION_BONUS_DEFAULT
    
    # 天气加成
    var weather_mult: float = WEATHER_BONUS_DEFAULT
    
    # 创意值加成
    var creativity_damage: float = creativity * skill_mult * STAT_MULTIPLIER
    
    # 最终伤害
    var raw_damage: float = (weapon_atk * weapon_mult + creativity_damage) * region_mult * weather_mult
    var final_damage: int = max(int(raw_damage - monster_def), 1)
    
    return final_damage

static func calc_enemy_damage(base_atk: int, defense: int, is_defending: bool) -> int:
    var raw := base_atk - defense
    if is_defending:
        raw = int(raw * 0.5)
    return max(raw, 1)

static func calc_counter_damage(enemy_base_atk: int, difficulty: int) -> int:
    var ratio: float = [0.3, 0.6, 1.0][clampi(difficulty - 1, 0, 2)]
    return int(enemy_base_atk * ratio * 1.5)  # 反伤倍率1.5
```

```gdscript
# scripts/scenes/qte_controller.gd
class_name QteController
extends Node

var question_pool: QTEQuestionPool = null
var current_question: QTEQuestion = null
var time_remaining: float = 5.0
var is_active: bool = false
var used_questions: Array[int] = []

signal qte_result(success: bool, counter_damage: float)

func _ready() -> void:
    question_pool = load("res://resources/qte/qte_question_pool.tres")

func start_qte(difficulty_min: int, difficulty_max: int) -> void:
    if question_pool == null or question_pool.questions.is_empty():
        qte_result.emit(false, 0.0)
        return
    
    # 按难度过滤
    var candidates: Array[QTEQuestion] = []
    for q in question_pool.questions:
        if q.difficulty >= difficulty_min and q.difficulty <= difficulty_max:
            if not used_questions.has(q.id):
                candidates.append(q)
    
    if candidates.is_empty():
        # 重置已用题库
        used_questions.clear()
        for q in question_pool.questions:
            if q.difficulty >= difficulty_min and q.difficulty <= difficulty_max:
                candidates.append(q)
    
    if candidates.is_empty():
        qte_result.emit(false, 0.0)
        return
    
    # 随机选取
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    current_question = candidates[rng.randi_range(0, candidates.size() - 1)]
    used_questions.append(current_question.id)
    
    time_remaining = 5.0
    is_active = true
    EventBus.qte_triggered.emit(current_question.id, current_question.difficulty)

func process_qte(delta: float) -> void:
    if not is_active:
        return
    time_remaining -= delta
    if time_remaining <= 0:
        _on_timeout()

func get_shuffled_options() -> Array:
    if current_question == null:
        return []
    var opts := current_question.options.duplicate()
    opts.shuffle()
    return opts

func get_correct_index(shuffled_options: Array) -> int:
    if current_question == null:
        return -1
    var correct_text := current_question.options[current_question.correct_index]
    return shuffled_options.find(correct_text)

func submit_answer(selected_index: int, shuffled_options: Array) -> void:
    if not is_active:
        return
    is_active = false
    
    var correct_idx := get_correct_index(shuffled_options)
    if selected_index == correct_idx:
        var counter_dmg := DamageCalculator.calc_counter_damage(50, current_question.difficulty)
        EventBus.qte_success.emit(counter_dmg)
        qte_result.emit(true, counter_dmg)
    else:
        var incoming_dmg := DamageCalculator.calc_enemy_damage(80, 10, false) * current_question.difficulty
        EventBus.qte_failed.emit(incoming_dmg)
        qte_result.emit(false, incoming_dmg)

func _on_timeout() -> void:
    is_active = false
    var incoming_dmg := DamageCalculator.calc_enemy_damage(80, 10, false) * current_question.difficulty
    EventBus.qte_failed.emit(incoming_dmg)
    qte_result.emit(false, incoming_dmg)
```

```gdscript
# scripts/scenes/enemy_ai.gd
class_name EnemyAI
extends RefCounted

func decide_action(monster: MonsterData, current_hp: int, max_hp: int, pending_charge: SkillData) -> Dictionary:
    var rng := RandomNumberGenerator.new()
    rng.randomize()
    
    # 正在蓄力中 → 释放蓄力
    if pending_charge != null:
        return {"action": "release_charge", "skill": pending_charge}
    
    var hp_ratio := float(current_hp) / float(max_hp)
    
    # 低血量 (< 30%) 概率使用蓄力
    if hp_ratio < 0.3 and monster.charge_skills.size() > 0:
        if rng.randf() < 0.5:
            var charge_skill := monster.charge_skills[rng.randi_range(0, monster.charge_skills.size() - 1)]
            return {"action": "charge", "skill": charge_skill}
    
    # 正常回合: 80%瞬发, 20%蓄力
    if monster.charge_skills.size() > 0 and rng.randf() < 0.2:
        var charge_skill := monster.charge_skills[rng.randi_range(0, monster.charge_skills.size() - 1)]
        return {"action": "charge", "skill": charge_skill}
    
    # 瞬发
    if monster.instant_skills.is_empty():
        return {"action": "instant", "skill": null}  # 普攻
    var skill := monster.instant_skills[rng.randi_range(0, monster.instant_skills.size() - 1)]
    return {"action": "instant", "skill": skill}
```
